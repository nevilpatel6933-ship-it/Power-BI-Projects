import json
import re
import threading
import time
from datetime import datetime
import pandas as pd

# ---------------- DECORATOR ----------------
def logger(func):
    def wrapper(*args, **kwargs):
        print(f"\n[LOG] Running: {func.__name__}")
        return func(*args, **kwargs)
    return wrapper


# ---------------- CLASS ----------------
class ExpenseManager:
    def __init__(self):
        self.expenses = []
        self.file = "expenses.json"
        self.limit = 5000
        self.load_expenses()

    # ---------------- FILE I/O ----------------
    def load_expenses(self):
        try:
            with open(self.file, "r") as f:
                self.expenses = json.load(f)
        except:
            self.expenses = []

    def save_expenses(self):
        with open(self.file, "w") as f:
            json.dump(self.expenses, f, indent=4)

    # ---------------- ADD EXPENSE ----------------
    @logger
    def add_expense(self):
        try:
            amount = float(input("Enter amount: "))

            category = input("Enter category: ")
            if not re.match(r"^[A-Za-z]+$", category):
                print("Invalid category!")
                return

            date = input("Enter date (YYYY-MM-DD): ")
            if not re.match(r"\d{4}-\d{2}-\d{2}", date):
                print("Invalid date format!")
                return

            expense = {
                "amount": amount,
                "category": category,
                "date": date
            }

            self.expenses.append(expense)
            self.save_expenses()

            print("✅ Expense added!")

        except ValueError:
            print("❌ Invalid amount!")

    # ---------------- VIEW ----------------
    @logger
    def view_expenses(self):
        for e in self.expenses:
            print(e)

    # ---------------- TOTAL ----------------
    @logger
    def total_expense(self):
        total = sum(map(lambda x: x["amount"], self.expenses))
        print("💰 Total Expense:", total)

    # ---------------- FILTER ----------------
    @logger
    def filter_category(self):
        cat = input("Enter category: ")
        filtered = list(filter(lambda x: x["category"] == cat, self.expenses))
        print(filtered)

    # ---------------- SORT ----------------
    @logger
    def sort_expenses(self):
        self.expenses.sort(key=lambda x: x["amount"])
        print("Sorted by amount!")

    # ---------------- REPORT (PANDAS) ----------------
    @logger
    def report(self):
        if not self.expenses:
            print("No data!")
            return

        df = pd.DataFrame(self.expenses)
        print("\n📊 Category-wise spending:")
        print(df.groupby("category")["amount"].sum())

    # ---------------- ALERT SYSTEM (THREAD) ----------------
    def alert_system(self):
        while True:
            total = sum(map(lambda x: x["amount"], self.expenses))
            if total > self.limit:
                print("\n⚠️ ALERT: Spending limit exceeded!")
            time.sleep(10)


# ---------------- MAIN ----------------
manager = ExpenseManager()

# Start background thread
thread = threading.Thread(target=manager.alert_system, daemon=True)
thread.start()

while True:
    print("\n===== EXPENSE TRACKER =====")
    print("1. Add Expense")
    print("2. View Expenses")
    print("3. Total Expense")
    print("4. Filter by Category")
    print("5. Sort Expenses")
    print("6. Generate Report")
    print("7. Exit")

    choice = input("Enter choice: ")

    if choice == "1":
        manager.add_expense()

    elif choice == "2":
        manager.view_expenses()

    elif choice == "3":
        manager.total_expense()

    elif choice == "4":
        manager.filter_category()

    elif choice == "5":
        manager.sort_expenses()

    elif choice == "6":
        manager.report()

    elif choice == "7":
        print("Exiting...")
        break

    else:
        print("Invalid choice!")
